cmake_minimum_required(VERSION 3.16)
project(clippy-ai-config LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
add_executable(clippy-ai-config main.cpp)
target_link_libraries(clippy-ai-config PRIVATE Qt6::Widgets Qt6::Network)
install(TARGETS clippy-ai-config RUNTIME DESTINATION bin)
